cmake_minimum_required(VERSION 3.22)
project(typelay_linux_native LANGUAGES C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GTK REQUIRED IMPORTED_TARGET gtk+-3.0 json-glib-1.0)
add_executable(typelay-supervise supervise.c)
add_library(typelay-gtk3 MODULE gtk-module.c gtk-target.c)
target_link_libraries(typelay-gtk3 PRIVATE PkgConfig::GTK)
foreach(target typelay-supervise typelay-gtk3)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -Wformat=2 -Wshadow -Wconversion -Wno-deprecated-declarations)
  target_link_options(${target} PRIVATE -Wl,-z,relro,-z,now -Wl,-z,noexecstack)
endforeach()
set_target_properties(typelay-gtk3 PROPERTIES C_VISIBILITY_PRESET hidden)

option(TYPELAY_NATIVE_TESTS "Build the isolated GTK contract fixture" OFF)
if(TYPELAY_NATIVE_TESTS)
  add_executable(gtk-fixture ../../../tests/linux/gtk_fixture.c)
  target_link_libraries(gtk-fixture PRIVATE PkgConfig::GTK)
  target_compile_options(gtk-fixture PRIVATE -Wall -Wextra -Werror)
endif()
