cmake_minimum_required(VERSION 3.22)
project(typelay_qt6_native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets)
add_library(typelay-qt6 MODULE qt-module.cpp)
target_link_libraries(typelay-qt6 PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets)
target_compile_options(typelay-qt6 PRIVATE -Wall -Wextra -Werror -Wformat=2 -Wshadow -Wconversion)
target_link_options(typelay-qt6 PRIVATE -Wl,-z,relro,-z,now -Wl,-z,noexecstack)
set_target_properties(typelay-qt6 PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/qt6/generic")
option(TYPELAY_QT_NATIVE_TESTS "Build the isolated QLineEdit native fixture" OFF)
if(TYPELAY_QT_NATIVE_TESTS)
  add_executable(qt-native-fixture ../../../../tests/linux/qt_native_fixture.cpp)
  target_link_libraries(qt-native-fixture PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets)
  target_compile_options(qt-native-fixture PRIVATE -Wall -Wextra -Werror)
endif()
