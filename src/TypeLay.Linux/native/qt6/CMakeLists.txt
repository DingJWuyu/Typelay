cmake_minimum_required(VERSION 3.22)
project(typelay_qt6 LANGUAGES CXX)
find_package(Qt6 6.2 REQUIRED COMPONENTS Core Gui Widgets Network)
add_library(typelay-qt6 MODULE typelay-qt6.cpp)
set_target_properties(typelay-qt6 PROPERTIES AUTOMOC ON CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/qt6/generic")
target_link_libraries(typelay-qt6 PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Network)
target_compile_options(typelay-qt6 PRIVATE -Wall -Wextra -Werror)
target_link_options(typelay-qt6 PRIVATE -Wl,-z,relro,-z,now -Wl,-z,noexecstack)

if(TYPELAY_NATIVE_TESTS)
  add_executable(qt-fixture ../../../../tests/linux/qt_fixture.cpp)
  target_link_libraries(qt-fixture PRIVATE Qt6::Widgets)
  set_target_properties(qt-fixture PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}" CXX_STANDARD 17)
  target_compile_options(qt-fixture PRIVATE -Wall -Wextra -Werror)
endif()
